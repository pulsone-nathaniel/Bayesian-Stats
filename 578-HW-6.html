# Initialize functions for drawing full conditionals

draw_mu = function(theta_vec, tao2){
  denom = (m/tao2 + 1/gam2_0)
  post_mean = (m*mean(theta_vec)/tao2 + mu_0/gam2_0)/denom
  post_sd = sqrt(1/denom)
  
  return(rnorm(1, post_mean, post_sd))
}

draw_tao2 = function(theta_vec, mu){
  post_shape = (eta_0+m)/2
  post_rate = (eta_0*tao2_0 + sum((theta_vec-mu)^2))/2
  
  return(1/rgamma(1, post_shape, post_rate))
}

draw_theta_vec = function(sig2, mu, tao2){
  theta_vec = c()
  for(j in 1:m){
    denom = n_vec[j]/sig2 + 1/tao2
    post_mean = (n_vec[j]*ybar_vec[j]/sig2 + mu/tao2)/denom
    post_sd = sqrt(1/denom)
    theta_vec = c(theta_vec, rnorm(1, post_mean, post_sd))
  }
  return(theta_vec)
}

draw_sig2 = function(theta_vec){
  ssr = 0
  
  for(j in 1:m){
    yj = as.numeric(ylist[[j]])
    ssr = ssr + sum((yj-theta_vec[j])^2)
    
  }
  post_shape = 0.5 * (nu_0 + sum(n_vec))
  post_rate = 0.5 * (nu_0*sig2_0 + ssr)
  sig2 = 1/rgamma(1, post_shape, post_rate)
  return(sig2)
}

# update functions
draw_betaj = function(yj, Xj, sig2, theta, Sigma_inv){
  varbj = solve(Sigma_inv + t(Xj)%*%Xj/sig2)
  ebj = varbj %*% Sigma_inv %*% theta + (t(Xj) %*% yj)/sig2
  return(mvrnorm(1, ebj, varbj))
}

draw_theta = function(betabar, Sigma_inv){
  Lambda_m = solve(Lambda_0 + m * Sigma_inv)
  mu_m = Lambda_m %*% (m * Sigma_inv %*% betabar)
  return(mvrnorm(1, mu_m, Lambda_m))
}

draw_Sigma_inv = function(Beta, theta){
  df = 4 + m
  S_theta = Sigma_hat_inv
  for(j in 1:m){
    betaj = Beta[j,]
    S_theta = S_theta + (betaj - theta)%*%t(betaj - theta)
  }
  
  return(rWishart(1, df, S_theta))
}

draw_sig2 = function(Beta){
  shape = 40.5
  ssr = 0
  for(j in 1:m){
    xj = xlist[[j]]
    Xj = cbind(1, xj, xj^2)
    ssr = ssr + sum((ylist[[j]] - Xj%*%Beta[j,])^2)
  }
  rate = (sig2_hat + ssr)/2
  return(1/rgamma(1, shape, rate))
  
}
